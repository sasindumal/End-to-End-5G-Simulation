\begin{lstlisting}[language=bash, title={Rebuild and run from a clean Mac}]
brew install lima                                  # once (requires accepted Xcode licence)
limactl create --name=e2e5g vm/lima-5g.yaml && limactl start e2e5g
limactl shell e2e5g -- bash /e2e5g/scripts/00_provision.sh
limactl shell e2e5g -- sudo bash -c 'cd /e2e5g/scripts && ./10_netns.sh && ./20_core.sh \
   && ./30_subscribers.sh && ./40_ran.sh && ./slice_ctl.sh init \
   && ../traffic/dn_servers.sh && ./60_sandbox.sh'      # then open http://localhost:8501
limactl shell e2e5g -- sudo /opt/e2e5g-venv/bin/python /e2e5g/experiments/run_all.py
limactl shell e2e5g -- /opt/e2e5g-venv/bin/python /e2e5g/analysis/analyze.py
\end{lstlisting}

\begin{lstlisting}[language=bash, title={scripts/slice\_ctl.sh (excerpt): live per-slice MBR and delay/loss}]
# downlink: HTB class + netem on the slice's UPF tun device
tc class change dev "$tun" parent 1: classid 1:10 htb rate "${rate}mbit" ceil "${rate}mbit" burst 64k
tc qdisc  change dev "$tun" parent 1:10 handle 10: netem delay ${delay}ms loss ${loss}% limit 10000
# uplink: one HTB class per UE pool on the N6 veth, selected by source subnet
tc filter add dev n6-core parent 1: protocol ip prio 1 u32 match ip src "$subnet" flowid 1:$cls
\end{lstlisting}
